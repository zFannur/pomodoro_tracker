```dart
/// Сводка одной недели (или любого набора дней — так переиспользуем для
/// разбиения по самочувствию).
class WeekInsight extends Equatable {
  final int activeDays;          // дни с ≥1 помидором
  final double? firstStart;      // минуты; см. §5
  final bool firstStartIsLag;    // true — опоздание к окну, false — минуты от 05:00
  final double? idle;
  final double? frogFirst;       // 0..100
  final int deferrals;           // эпизоды source == defer (frog не считать — это отдельный сигнал)
  final double? distract;        // 0..100, приходит снаружи
  final int episodes;
  final Map<Cause, int> causes;
  final Map<StrategyKind, ({int tries, int wins})> strategies;
}

WeekInsight weekInsight(
  List<DayLog> days, {
  List<SessionWindow> windows = const [],
  double? distractPercent,
});

bool episodeSucceeded(Episode e, List<PomoSession> sessions);
Map<StrategyKind, ({int tries, int wins})> strategyStats(List<DayLog> days);
double? metricValue(WeekInsight w, InsightMetric m);   // success = Σwins/Σtries·100
bool lowerIsBetter(InsightMetric m);                    // firstStart, idle, deferrals, distract
Verdict verdictFor({required InsightMetric metric, double? baseline, double? now, required int days});
String? baselineWeek(List<DayLog> days);                // sprintId самой ранней недели с эпизодом
bool isObservationWeek(String currentWeek, String? baseline); // baseline == null || == current

enum CheckInField { sleep, energy }
class CheckInSplit { final double? good; final double? bad; final int goodDays; final int badDays; }
CheckInSplit? checkInSplit(List<DayLog> days, CheckInField f, InsightMetric m, {List<SessionWindow> windows = const []});
```

Правила (точные):
- **Успех эпизода**: есть сессия `s`, у которой `!s.manual`, `s.minutes >= 5`,
  `s.start < e.at + 15 мин` и `s.start + s.minutes > e.at`. Сессии брать того же
  логического дня (вызывающий передаёт сессии дня эпизода).
- **firstStart**: дни с ≥1 не-ручной сессией; первая по `start`. Если `windows`
  непуст — `max(0, start − начало первого окна того дня)` в минутах,
  `firstStartIsLag = true`; иначе минуты от 05:00 логического дня. Дней < 3 → null.
- **idle**: среднее `delayMinutes` по не-ручным сессиям, кроме первой сессии
  каждого дня. Сессий < 5 → null.
- **frogFirst**: доля активных дней, где первая сессия — `frog`, ×100. Активных < 3 → null.
- **causes**: по эпизодам с `cause != null`, включая `external` (UI покажет его отдельно).
- **strategies**: попытка — эпизод со `strategy != null`; победа — `episodeSucceeded`.
- **verdictFor**: `days < 3` или `baseline == null` или `now == null` → `unclear`.
  Если `baseline == 0`: `now == 0` → `same`; иначе `worse` для lowerIsBetter, `works`
  для остальных. Иначе `delta = (now − baseline) / baseline`; для lowerIsBetter
  знак инвертировать; `delta ≥ 0.15` → `works`, `≤ −0.15` → `worse`, иначе `same`.
- **checkInSplit**: good — дни с полем ≥4, bad — с полем ≤2 (3 не участвует). Любая
  группа < 3 дней → null. Значения — `metricValue(weekInsight(группа), m)`.
